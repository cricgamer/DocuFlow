/** * DocuFlow Studio - Main Application Controller * Handles routing, sidebar navigation, keyboard shortcuts, and initialization */ window.DocuApp = (function () { 'use strict'; let currentTab = 'editor'; // 'editor' | 'merge' | 'organize' | 'watermark' function init() { // 1. Initialize Submodules if (window.DocuEditor) DocuEditor.init(); if (window.DocuMerge) DocuMerge.init(); if (window.DocuOrganize) DocuOrganize.init(); if (window.DocuWatermark) DocuWatermark.init(); // 2. Setup Navigation setupNavigation(); // 3. Setup Mobile Drawer setupMobileSidebar(); // 4. Setup Keyboard Shortcuts setupKeyboardShortcuts(); // 5. Initialize Lucide Icons refreshIcons(); // Welcome toast setTimeout(() => { DocuUtils.showToast( 'DocuFlow Studio Ready', '100% Client-Side Private Document Workspace. Zero data leaves your machine.', 'info', 4000 ); }, 400); } function setupNavigation() { const navButtons = document.querySelectorAll('[data-nav-target]'); navButtons.forEach(btn => { btn.addEventListener('click', (e) => { e.preventDefault(); const target = btn.getAttribute('data-nav-target'); switchTab(target); }); }); } function switchTab(tabId) { if (tabId === currentTab) return; // Hide all tab panels document.querySelectorAll('.tab-panel').forEach(panel => { panel.classList.add('hidden'); }); // Show target panel const targetPanel = document.getElementById(`tab-content-${tabId}`); if (targetPanel) { targetPanel.classList.remove('hidden'); } // Update active nav styling document.querySelectorAll('[data-nav-target]').forEach(btn => { const isTarget = btn.getAttribute('data-nav-target') === tabId; if (isTarget) { btn.classList.add('bg-sky-500/10', 'text-sky-400', 'border-sky-500/40'); btn.classList.remove('text-slate-400', 'hover:bg-slate-800/60', 'border-transparent'); } else { btn.classList.remove('bg-sky-500/10', 'text-sky-400', 'border-sky-500/40'); btn.classList.add('text-slate-400', 'hover:bg-slate-800/60', 'border-transparent'); } }); currentTab = tabId; // Update Header Label & Sample Button const headerLabel = document.getElementById('header-workspace-label'); const headerSampleBtn = document.getElementById('header-sample-btn'); const meta = { editor: { label: 'editor', sampleText: 'Sample Tech Spec', sampleAction: () => DocuEditor.loadTemplate(DocuEditor.TEMPLATES ? DocuEditor.TEMPLATES.techSpec : '', 'Tech-Spec') }, merge: { label: 'pdf-merge', sampleText: 'Load 2 Demo PDFs', sampleAction: () => DocuMerge.loadSamplePdf() }, organize: { label: 'pdf-organize', sampleText: 'Load 4-Page Demo PDF', sampleAction: () => DocuOrganize.loadSamplePdf() }, watermark: { label: 'pdf-watermark', sampleText: 'Load Sample NDA', sampleAction: () => DocuWatermark.loadSamplePdf() } }; if (meta[tabId]) { if (headerLabel) headerLabel.textContent = meta[tabId].label; if (headerSampleBtn) { headerSampleBtn.innerHTML = ` ${meta[tabId].sampleText} `; headerSampleBtn.onclick = meta[tabId].sampleAction; } } // Close mobile drawer if open closeMobileSidebar(); // Re-render icons if needed refreshIcons(); } function setupMobileSidebar() { const toggleBtn = document.getElementById('mobile-sidebar-toggle'); const sidebar = document.getElementById('app-sidebar'); const backdrop = document.getElementById('sidebar-backdrop'); if (!toggleBtn || !sidebar) return; toggleBtn.addEventListener('click', () => { sidebar.classList.toggle('-translate-x-full'); if (backdrop) backdrop.classList.toggle('hidden'); }); if (backdrop) { backdrop.addEventListener('click', closeMobileSidebar); } } function closeMobileSidebar() { const sidebar = document.getElementById('app-sidebar'); const backdrop = document.getElementById('sidebar-backdrop'); if (sidebar && !sidebar.classList.contains('-translate-x-full')) { sidebar.classList.add('-translate-x-full'); } if (backdrop) backdrop.classList.add('hidden'); } function setupKeyboardShortcuts() { window.addEventListener('keydown', (e) => { // Ctrl+S or Cmd+S to export current document if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 's') { e.preventDefault(); if (currentTab === 'editor') { DocuEditor.exportAsDocx(); } else if (currentTab === 'merge') { DocuMerge.executeMerge(); } else if (currentTab === 'organize') { DocuOrganize.saveOrganizedPdf(); } else if (currentTab === 'watermark') { DocuWatermark.applyWatermark(); } } }); } function refreshIcons() { if (window.lucide && typeof window.lucide.createIcons === 'function') { window.lucide.createIcons(); } } function openPrivacyModal() { const modal = document.getElementById('privacy-modal'); if (modal) { modal.classList.remove('hidden'); modal.classList.add('flex'); } } function closePrivacyModal() { const modal = document.getElementById('privacy-modal'); if (modal) { modal.classList.add('hidden'); modal.classList.remove('flex'); } } return { init, switchTab, openPrivacyModal, closePrivacyModal, refreshIcons }; })(); // Bootstrap on DOM ready if (typeof document !== 'undefined' && document.addEventListener) { document.addEventListener('DOMContentLoaded', () => { DocuApp.init(); }); }
